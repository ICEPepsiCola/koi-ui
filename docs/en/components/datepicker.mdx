# DatePicker

## Basic usage

```tsx preview
import { DatePicker } from '@koi-ui/core';
import { useState } from 'react';
export default () => {
  const [v, setV] = useState('');
  return (
    <div className="w-full max-w-xs">
      <DatePicker value={v} onChange={setV} placeholder="Select date" />
    </div>
  );
};
```

## Controlled date

```tsx preview
import { DatePicker, Text } from '@koi-ui/core';
import { useState } from 'react';
export default () => {
  const [date, setDate] = useState('2026-07-12');
  return (
    <div className="w-full max-w-xs">
      <DatePicker value={date} onChange={setDate} placeholder="Select date" />
      <Text muted className="mt-2 block text-sm">Selected: {date || 'None'}</Text>
    </div>
  );
};
```

## Year / month / week

Use `picker` for granularity. Week values are ISO weeks: `YYYY-Www`.

```tsx preview
import { DatePicker, Space } from '@koi-ui/core';
import { useState } from 'react';
export default () => {
  const [year, setYear] = useState('2026');
  const [month, setMonth] = useState('2026-07');
  const [week, setWeek] = useState('2026-W28');
  return (
    <Space direction="vertical" size="middle" className="w-full max-w-xs">
      <DatePicker picker="year" value={year} onChange={setYear} placeholder="Year" />
      <DatePicker picker="month" value={month} onChange={setMonth} placeholder="Month" />
      <DatePicker picker="week" value={week} onChange={setWeek} placeholder="Week" />
    </Space>
  );
};
```

## Disabled dates

`disabledDate` matches Calendar: dates returning `true` are grayed out and not selectable (also respects `min` / `max`).

```tsx preview
import { DatePicker } from '@koi-ui/core';
import { useState } from 'react';

/** Example: only Tue / Fri (draw days) */
function isDrawDay(date: Date) {
  const day = date.getDay();
  return day !== 2 && day !== 5;
}

export default () => {
  const [v, setV] = useState('');
  return (
    <div className="w-full max-w-xs">
      <DatePicker
        value={v}
        onChange={setV}
        disabledDate={isDrawDay}
        clearable
        placeholder="Draw days only"
      />
    </div>
  );
};
```

## Date range

Use `range`, or the Ant-style `DatePicker.RangePicker` subcomponent (tuple-typed value).  
Desktop shows a **dual-month** panel; `placeholder` accepts `[start, end]`.

```tsx preview
import { DatePicker, Text, Space } from '@koi-ui/core';
import { useState } from 'react';
export default () => {
  const [range, setRange] = useState<[string, string]>(['', '']);
  return (
    <Space direction="vertical" size="middle" className="w-full max-w-xl">
      <DatePicker.RangePicker
        value={range}
        onChange={setRange}
        clearable
        placeholder={['Start date', 'End date']}
      />
      <Text muted className="text-sm">
        {range[0] && range[1] ? `${range[0]} ~ ${range[1]}` : 'None'}
      </Text>
    </Space>
  );
};
```
